        <button onclick="searchDomain()">Search</button>

    <!-- Iframe Container -->
    <iframe id="crtIframe" src="about:blank"></iframe>

    <script>
        function searchDomain() {
            let domain = document.getElementById('domainInput').value.trim();
            
            if (domain) {
                // protocol, http/https বা www থাকলে তা রিমুভ করা
                domain = domain.replace(/^(?:https?:\/\/)?(?:www\.)?/i, "");
                
                // Target URL তৈরি
                const targetUrl = `https://crt.name/v1/search?apex=${encodeURIComponent(domain)}`;
                
                // Iframe-এ URL সেট করা
                document.getElementById('crtIframe').src = targetUrl;
            } else {
                alert('অনুগ্রহ করে একটি ডোমেইন ইনপুট দিন।');
            }
        }

        // Enter কী প্রেস করলে সার্চ হবে
        document.getElementById('domainInput').addEventListener('keypress', function (e) {
            if (e.key === 'Enter') {
                searchDomain();
            }
        });
    </script>
